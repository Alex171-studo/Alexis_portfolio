Les commandes à retenir 

man <commande> : Afficher le manuel de la commande 

whoami : afficher le nom d'utilisateur actuel

id : affichie l'identité de l'utilisateur (son id, guid , etc)

hostame : afficher le nom de l'ordinateur

uname -a : affiche les infos sur le sytème d'exploitation

cat /etc/os-release : savoir quelle version d'OS

pwd : affiche le répertoire actuel 

who : affiche qui est connecté

lsblk : affiche tous les périphériques de stockage connectés

lsusb : affiche tous les périphériques USB connectés

lspci : affiche tous les périphériques connectées au us CPi

lsof : affiche tous les fichiers ouverts par le processus

which <commande> : l'exécutable du fichier

find -size +x -a -y : taille compris entre x et y 

find -newermt date : fichier après cette date 

ps -eo ( e pour tous les process et à o pour imposer un champ %cpu,comm)

exec > >(tee >file.txt) : rediriger toute les commandes suivantes dans un fichier

# FILTERING 
grep "error" file.log          # Chercher "error"
grep -i "error" file.log       # Case insensitive
grep -r "password" /var/log    # Récursif
grep -v "info" file.log        # Inverse (tout sauf info)
grep -c "failed" file.log      # Compter occurrences

cat file.log | grep "error" | cut -d' ' -f1,5  # Extraire colonnes
sort file.txt | uniq           # Dédupliquer
wc -l file.txt                 # Compter lignes

# REGEX
grep -E '\b[0-9]{1,3}(\.[0-9]{1,3}){3}\b' file.log  # IPs
grep -E '[a-z0-9]+@[a-z]+\.[a-z]+' file.txt         # Emails

# PERMISSIONS
chmod 755 script.sh            # rwxr-xr-x
chmod +x script.sh             # Rendre exécutable
chown user:group file          # Changer propriétaire
ls -la                         # Voir permissions

##PLANNIFICATION DES TÂCHES 

	#SYSTEMD
	
		-Créer le timer(dossier devant le contenir)
			sudo mkdir /etc/systemd/system/mytimer.timer.d
			
		-Coder le script du timer ( décide quand on exécute)
			sudo vim /etc/systemd/system/mytimer.timer
			
			[Unit] ##Description
			Description=My Timer

			[Timer] ##Action a exécuter
			OnBootSec=3min
			OnUnitActiveSec=1hour

			[Install] ##Où installer le timer
			WantedBy=timers.target
			
		-Coder le script à lancer par le timer (définis ce qui est exécuté)
			sudo vim /etc/systemd/system/mytimer.service
			
			[Unit]
			Description=My Service

			[Service]
			ExecStart=/full/path/to/my/script.sh

			[Install]
			WantedBy=multi-user.target
			
		- Recahrger systemd
			sudo systemctl daemon-reload
		
		-Lancer et activer le service timer
			sudo systemctl start mytimer.timer
			systemctl enable mytimer.timer

##Network
Network access control
DAC: le propriétaire décide de qui à accès à ses ressources
MAC : permissions renforcés par l'OS mais moins flexible. Chaque user et files ont un level de sécurité et l'accès est basé dessus
RBAC: les permissions sont définies en fonction du rôle

ip a or ifconfig : voire la config réseau
sudo ip link set eth0 up or sudo ifconfig eth0 up : activer l'interface eth0
sudo ip link addr add 192.168.1.12 dev eth0 or sudo ifconfig eth0 192.168.1.12 : atribuer une ip 
sudo ip link addr add 255.255.255.0/24 dev eth0 or sudo ifconfig eth0 netmask 255.255.255.0 : définir le masque
sudo ip route add default via 192.168.1.1 dev eth0 or sudo route add default gw 192.168.1.1 eth0 : gateway
cat /etc/resolv.conf : définir dans l'ordre les serveurs dns

##IPATBLES
-p or --protocol	Specifies the protocol to match (e.g. tcp, udp, icmp)
--dport	Specifies the destination port to match
--sport	Specifies the source port to match
-s or --source	Specifies the source IP address to match
-d or --destination	Specifies the destination IP address to match
-m state	Matches the state of a connection (e.g. NEW, ESTABLISHED, RELATED)
-m multiport	Matches multiple ports or port ranges
-m tcp	Matches TCP packets and includes additional TCP-specific options
-m udp	Matches UDP packets and includes additional UDP-specific options
-m string	Matches packets that contain a specific string
-m limit	Matches packets at a specified rate limit
-m conntrack	Matches packets based on their connection tracking information
-m mark	Matches packets based on their Netfilter mark value
-m mac	Matches packets based on their MAC address
-m iprange	Matches packets based on a range of IP addresses
