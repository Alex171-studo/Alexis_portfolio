Les commandes à retenir 

man <commande> : Afficher le manuel de la commande 

whoami : afficher le nom d'utilisateur actuel

id : affichie l'identité de l'utilisateur (son id, guid , etc)

hostame : afficher le nom de l'ordinateur

uname -a : affiche les infos sur le sytème d'exploitation

pwd : affiche le répertoire actuel 

who : affiche qui est connecté

lsblk : affiche tous les périphériques de stockage connectés

lsusb : affiche tous les périphériques USB connectés

lspci : affiche tous les périphériques connectées au us CPi

lsof : affiche tous les fichiers ouverts par le processus

which <commande> : l'exécutable du fichier

find -size +x -a -y : taille compris entre x et y 

find -newermt date : fichier après cette date 

# FILTERING 
grep "error" file.log          # Chercher "error"
grep -i "error" file.log       # Case insensitive
grep -r "password" /var/log    # Récursif
grep -v "info" file.log        # Inverse (tout sauf info)
grep -c "failed" file.log      # Compter occurrences

cat file.log | grep "error" | cut -d' ' -f1,5  # Extraire colonnes
sort file.txt | uniq           # Dédupliquer
wc -l file.txt                 # Compter lignes

# REGEX
grep -E '\b[0-9]{1,3}(\.[0-9]{1,3}){3}\b' file.log  # IPs
grep -E '[a-z0-9]+@[a-z]+\.[a-z]+' file.txt         # Emails

# PERMISSIONS
chmod 755 script.sh            # rwxr-xr-x
chmod +x script.sh             # Rendre exécutable
chown user:group file          # Changer propriétaire
ls -la                         # Voir permissions

# GESTION DES PAQUETS	

apt list --installed          #Voire la liste des paquets installés
dpkg -i nom_paquet.deb        #Installer un paquet

#SERVICES

systemctl start <service>    #Démarer un service
systemctl status <service>   #Etat du service
systemctl enable <service>   #Activer un service
systemctl list-units --type=service  #Lister tous les services
journactl -u name.service --no-pager  #Afficher les logs sur un service

#PROCESSUS

kill -l                      # Liste des signaux


##PLANNIFICATION DES TÂCHES 

	#SYSTEMD
	
		-Créer le timer(dossier devant le contenir)
			sudo mkdir /etc/systemd/system/mytimer.timer.d
			
		-Coder le script du timer ( décide quand on exécute)
			sudo vim /etc/systemd/system/mytimer.timer
			
			[Unit] ##Description
			Description=My Timer

			[Timer] ##Action a exécuter
			OnBootSec=3min
			OnUnitActiveSec=1hour

			[Install] ##Où installer le timer
			WantedBy=timers.target
			
		-Coder le script à lancer par le timer (définis ce qui est exécuté)
			sudo vim /etc/systemd/system/mytimer.service
			
			[Unit]
			Description=My Service

			[Service]
			ExecStart=/full/path/to/my/script.sh

			[Install]
			WantedBy=multi-user.target
			
		- Recahrger systemd
			sudo systemctl daemon-reload
		
		-Lancer et activer le service timer
			sudo systemctl start mytimer.timer
			systemctl enable mytimer.timer

	rsync -avz -e ssh <path_to_directory> user@server:<path_to_remote_directory>   # pour sécuriser la sauvegarde avec ssh
